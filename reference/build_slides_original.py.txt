from pptx.enum.shapes import MSO_SHAPE, MSO_CONNECTOR

BASE=Path(__file__).resolve().parent
OUT=BASE/'Review_Presentation_19_Nov_2026.pptx'
R=BASE/'results'
cfg=json.loads((BASE/'config.json').read_text())
def load(p):
 with (R/p).open(encoding='utf-8-sig',newline='') as f:return list(csv.DictReader(f))

prs=Presentation(); prs.slide_width=Inches(13.333);prs.slide_height=Inches(7.5)
blank=prs.slide_layouts[6]

def shape(sl,x,y,w,h,color,radius=True,line=None):
 k=sl.shapes.add_shape(MSO_SHAPE.ROUNDED_RECTANGLE if radius else MSO_SHAPE.RECTANGLE,Inches(x),Inches(y),Inches(w),Inches(h))
 k.fill.solid();k.fill.fore_color.rgb=color
 k.line.fill.background() if line is None else None
 if line is not None:k.line.color.rgb=line
 return k

def tx(sl,content,x,y,w,h,size=20,color=txt,bold=False,align=None,font='Aptos',margin=0.02):
 box=sl.shapes.add_textbox(Inches(x),Inches(y),Inches(w),Inches(h))
 tf=box.text_frame;tf.clear();tf.word_wrap=True
 tf.margin_left=Inches(margin);tf.margin_right=Inches(margin);tf.margin_top=Inches(0.01)
 for j,line in enumerate(str(content).split('\n')):
  p=tf.paragraphs[0] if j==0 else tf.add_paragraph()
  p.text=line;p.font.name=font;p.font.size=Pt(size);p.font.bold=bold;p.font.color.rgb=color
  p.space_after=Pt(6);p.alignment=align or PP_ALIGN.LEFT
 return box

def base(title,section='',num=0):
 sl=prs.slides.add_slide(blank)
 shape(sl,0,0,13.333,7.5,off,False)
 shape(sl,0,0,13.333,.16,teal,False)
 tx(sl,section.upper(),.55,.35,7,.27,12,teal,True)
 tx(sl,title,.55,.83,12.35,.69,27,navy,True)
 tx(sl,'FINAL REVIEW  •  19 NOV 2026  •  SIMULATION-BASED PROTOTYPE',.55,7.08,10,.2,9,mute)
 tx(sl,f'{num:02d}',12.25,7.04,.5,.22,11,mute,True,PP_ALIGN.RIGHT)
 return sl

def bullet(sl,items,x,y,w,size=18,lineheight=.65):
 for idx,item in enumerate(items):
  shape(sl,x,y+idx*lineheight+.09,.11,.11,teal)
  tx(sl,item,x+.25,y+idx*lineheight,w-.25,lineheight-.05,size,txt)

def image(sl,file,x,y,w,h):
 sl.shapes.add_picture(str(R/file),Inches(x),Inches(y),width=Inches(w),height=Inches(h))

def bottom_note(sl,msg):
 shape(sl,.55,6.50,12.22,.43,RGBColor(255,244,223))
 tx(sl,msg,.70,6.59,11.8,.26,12,RGBColor(115,77,32),False)

def summary_table(sl,x,y,w,h):
 headers=['Policy','Accuracy','Time (s)','Energy (Wh)','CO₂ (g)','Hit rate']
 t=sl.shapes.add_table(len(summ)+1,len(headers),Inches(x),Inches(y),Inches(w),Inches(h)).table
 t.columns[0].width=Inches(3.12)
 for j,head in enumerate(headers):t.cell(0,j).text=head
 for i,row in enumerate(summ,1):
  vals=[row['policy'],f"{float(row['final_accuracy_mean'])*100:.2f}%",row['mean_total_seconds_MODELED'],row['mean_total_Wh_MODELED'],row['mean_total_CO2_g_MODELED'],row['target_hit_rate']]
  for j,v in enumerate(vals):t.cell(i,j).text=str(v)
 for r in range(len(summ)+1):
  for c in range(len(headers)):
   cell=t.cell(r,c);cell.fill.solid();cell.fill.fore_color.rgb=navy if r==0 else white if r%2 else RGBColor(235,243,250)
   cell.vertical_anchor=MSO_ANCHOR.MIDDLE
   for p in cell.text_frame.paragraphs:
    p.font.name='Aptos';p.font.size=Pt(13 if c else 13.5);p.font.bold=(r==0);p.font.color.rgb=white if r==0 else txt
